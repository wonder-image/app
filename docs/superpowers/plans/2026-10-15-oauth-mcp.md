```php
<?php

namespace Wonder\OAuth;

use RuntimeException;

/** Errore OAuth nel formato RFC 6749: `error` + `error_description`, mai dettagli interni. */
final class OAuthException extends RuntimeException
{
    public function __construct(
        public readonly string $error,
        public readonly string $description = '',
        public readonly int $status = 400,
        public readonly array $headers = [],
        public readonly ?string $redirectUri = null,
        public readonly ?string $state = null,
    ) {
        parent::__construct($description !== '' ? $description : $error);
    }

    public function toArray(): array
    {
        return array_filter([
            'error' => $this->error,
            'error_description' => $this->description,
        ], static fn ($v) => $v !== '');
    }

    public function send(): void
    {
        if (!headers_sent()) {
            http_response_code($this->status);
            Http::noStore();
            header('Content-Type: application/json; charset=utf-8');
            foreach ($this->headers as $name => $value) {
                header($name.': '.$value);
            }
        }
        echo json_encode($this->toArray(), JSON_UNESCAPED_SLASHES);
    }
}
```

```php
<?php

namespace Wonder\OAuth;

use Wonder\App\Credentials;
use Wonder\App\Environment;

/** Valori OAuth modificabili da `.env`, con default e limiti (vedi spec, "Configurazione"). */
final class Config
{
    public const CODE_TTL = 300;
    public const REQUEST_TTL = 600;
    public const MAX_REQUESTS = 10;

    /** chiave => [default, minimo, massimo] */
    private const KEYS = [
        'OAUTH_REGISTER_RATE_LIMIT' => [20, 0, 1000],
        'OAUTH_ACCESS_TOKEN_TTL' => [3600, 300, 86400],
        'OAUTH_REFRESH_TOKEN_TTL' => [2592000, 3600, 7776000],
        'OAUTH_GRANT_MAX_AGE' => [7776000, 86400, 31536000],
    ];

    private static array $warned = [];

    public static function registerRateLimit(): int { return self::int('OAUTH_REGISTER_RATE_LIMIT'); }
    public static function accessTokenTtl(): int { return self::int('OAUTH_ACCESS_TOKEN_TTL'); }
    public static function refreshTokenTtl(): int { return self::int('OAUTH_REFRESH_TOKEN_TTL'); }
    public static function grantMaxAge(): int { return self::int('OAUTH_GRANT_MAX_AGE'); }

    public static function resetWarnings(): void { self::$warned = []; }

    /** Issuer = APP_URL senza slash finale. Mai l'header Host. */
    public static function issuer(): string
    {
        $url = rtrim(trim((string) (self::raw('APP_URL') ?? '')), '/');

        if ($url === '' || (!str_starts_with($url, 'https://') && !Environment::isLocal())) {
            error_log('[oauth] APP_URL deve essere https fuori dall\'ambiente locale: endpoint OAuth disattivati.');
            throw new OAuthException('server_error', 'Authorization server misconfigured.', 500);
        }

        return $url;
    }

    private static function int(string $key): int
    {
        [$default, $min, $max] = self::KEYS[$key];
        $raw = self::raw($key);
        $raw = is_string($raw) ? trim($raw) : $raw;

        if (!is_string($raw) || $raw === '' || preg_match('/^-?\d+$/', $raw) !== 1) {
            return $default;
        }

        $value = (int) $raw;
        $clamped = max($min, min($max, $value));

        if ($clamped !== $value && !isset(self::$warned[$key])) {
            self::$warned[$key] = true;
            error_log("[oauth] {$key}={$value} fuori dai limiti {$min}-{$max}: uso {$clamped}.");
        }

        return $clamped;
    }

    private static function raw(string $key): mixed
    {
        if (class_exists(Credentials::class)) {
            Credentials::loadEnv();
        }

        $value = $_ENV[$key] ?? $_SERVER[$key] ?? getenv($key);

        return $value === false ? null : $value;
    }
}
```

```php
<?php

namespace Wonder\OAuth;

/** PKCE solo S256 (RFC 7636). */
final class Pkce
{
    public static function validVerifier(string $verifier): bool
    {
        return preg_match('/^[A-Za-z0-9\-._~]{43,128}$/', $verifier) === 1;
    }

    public static function validChallenge(string $challenge): bool
    {
        return preg_match('/^[A-Za-z0-9_-]{43}$/', $challenge) === 1;
    }

    public static function challenge(string $verifier): string
    {
        return rtrim(strtr(base64_encode(hash('sha256', $verifier, true)), '+/', '-_'), '=');
    }

    public static function verify(string $verifier, string $challenge): bool
    {
        return self::validVerifier($verifier) && hash_equals($challenge, self::challenge($verifier));
    }
}
```

```php
<?php

namespace Wonder\OAuth;

/** Redirect URI: https o loopback; confronto esatto, porta ignorata solo sul loopback (RFC 8252 §7.3). */
final class RedirectUri
{
    private const LOOPBACK = ['localhost', '127.0.0.1', '[::1]'];

    public static function isAllowed(string $uri): bool
    {
        if ($uri === '' || strlen($uri) > 2000 || str_contains($uri, '#')) {
            return false;
        }

        $parts = parse_url($uri);

        if (!is_array($parts) || empty($parts['scheme']) || empty($parts['host'])
            || isset($parts['user']) || isset($parts['pass'])) {
            return false;
        }

        $scheme = strtolower($parts['scheme']);

        return $scheme === 'https' || ($scheme === 'http' && self::isLoopback($uri));
    }

    public static function isLoopback(string $uri): bool
    {
        $parts = parse_url($uri);

        return is_array($parts)
            && strtolower((string) ($parts['scheme'] ?? '')) === 'http'
            && in_array(strtolower((string) ($parts['host'] ?? '')), self::LOOPBACK, true);
    }

    public static function matches(string $registered, string $presented): bool
    {
        if (!self::isLoopback($registered) || !self::isLoopback($presented)) {
            return hash_equals($registered, $presented);
        }

        return hash_equals(self::withoutPort($registered), self::withoutPort($presented));
    }

    public static function matchesAny(array $registered, string $presented): bool
    {
        foreach ($registered as $uri) {
            if (is_string($uri) && self::matches($uri, $presented)) {
                return true;
            }
        }

        return false;
    }

    public static function host(string $uri): string
    {
        return strtolower((string) (parse_url($uri, PHP_URL_HOST) ?? ''));
    }

    private static function withoutPort(string $uri): string
    {
        $p = parse_url($uri);

        return strtolower((string) $p['scheme']).'://'.strtolower((string) $p['host'])
```

```php
namespace Wonder\OAuth\Store;

/** Persistenza in memoria per i test: stesso contratto di SqlStore. */
final class MemoryStore implements OAuthStore
{
    public array $tables = ['clients' => [], 'grants' => [], 'access' => [], 'refresh' => [], 'codes' => []];
    public array $users = [];
    private int $seq = 0;

    public function now(): string { return date('Y-m-d H:i:s'); }

    public function createClient(array $row): void
    {
        $this->tables['clients'][$row['client_id']] = $row + ['id' => ++$this->seq, 'last_used_at' => null, 'revoked_at' => null, 'deleted' => 'false', 'client_secret_hash' => null];
    }

    public function findClient(string $clientId): ?array
    {
        $row = $this->tables['clients'][$clientId] ?? null;
        return ($row === null || $row['deleted'] === 'true') ? null : $row;
    }

    public function countClientsFromIp(string $ip, string $since): int
    {
        return count(array_filter($this->tables['clients'], static fn ($c) => $c['registration_ip'] === $ip && $c['creation'] >= $since));
    }

    public function touchClient(string $clientId, string $now): void
    {
        if (isset($this->tables['clients'][$clientId])) {
            $this->tables['clients'][$clientId]['last_used_at'] = $now;
        }
    }

    public function deleteClient(string $clientId): void { unset($this->tables['clients'][$clientId]); }

    public function createGrant(array $row): int
    {
        $id = ++$this->seq;
        $this->tables['grants'][$id] = $row + ['id' => $id, 'last_used_at' => null, 'revoked_at' => null];
        return $id;
    }

    public function findGrant(int $id): ?array { return $this->tables['grants'][$id] ?? null; }

    public function findActiveGrant(int $userId, string $clientId, string $resource): ?array
    {
        foreach ($this->tables['grants'] as $g) {
            if ((int) $g['user_id'] === $userId && $g['client_id'] === $clientId && $g['resource'] === $resource && $g['revoked_at'] === null) {
                return $g;
            }
        }
        return null;
    }

    public function updateGrant(int $id, array $values): void
    {
        if (isset($this->tables['grants'][$id])) {
            $this->tables['grants'][$id] = $values + $this->tables['grants'][$id];
        }
    }

    public function revokeGrant(int $id, string $now): void
    {
        if (isset($this->tables['grants'][$id]) && $this->tables['grants'][$id]['revoked_at'] === null) {
            $this->tables['grants'][$id]['revoked_at'] = $now;
        }
        foreach (['access', 'refresh'] as $kind) {
            foreach ($this->tables[$kind] as $sel => $t) {
                if ((int) $t['grant_id'] === $id && $t['revoked_at'] === null) {
                    $this->tables[$kind][$sel]['revoked_at'] = $now;
                }
            }
        }
    }

    public function listActiveGrants(?int $userId): array
    {
        $rows = array_values(array_filter($this->tables['grants'], static fn ($g) => $g['revoked_at'] === null && ($userId === null || (int) $g['user_id'] === $userId)));
        usort($rows, static fn ($a, $b) => strcmp($b['consented_at'], $a['consented_at']));

        $this->tables[$kind][$row['selector']] = $row + ['id' => ++$this->seq, 'revoked_at' => null, 'consumed_at' => null];
    }

    public function findToken(string $kind, string $selector): ?array { return $this->tables[$kind][$selector] ?? null; }

    public function consumeRefresh(int $id, string $now): bool
    {
        foreach ($this->tables['refresh'] as $sel => $t) {
            if ($t['id'] === $id) {
                if ($t['consumed_at'] !== null || $t['revoked_at'] !== null) {
                    return false;
                }
                $this->tables['refresh'][$sel]['consumed_at'] = $now;
                return true;
            }
        }
        return false;

        $this->tables['codes'][$code] = ['user_id' => $userId, 'metadata' => $metadata, 'expires' => time() + $ttl, 'used' => false];
        return $code;
    }

    public function consumeCode(string $code): ?array
    {
        $row = $this->tables['codes'][$code] ?? null;
        if ($row === null || $row['used'] || $row['expires'] < time()) {
            return null;
        }
        $this->tables['codes'][$code]['used'] = true;
        return ['user_id' => $row['user_id'], 'metadata' => $row['metadata']];
    }

    public function findUser(int $id): ?object { return $this->users[$id] ?? null; }

    public function purgeTokens(string $before): int

        foreach (['access', 'refresh'] as $kind) {
            foreach ($this->tables[$kind] as $sel => $t) {
                $dead = $t['expires_at'] < $before
                    || (($t['consumed_at'] ?? null) !== null && $t['consumed_at'] < $before)

            }
        }
        return $n;
    }

        foreach ($this->tables['clients'] as $id => $c) {
            $hasActive = false;
            foreach ($this->tables['grants'] as $g) {

            if (!$hasActive && $last < $idleBefore) { unset($this->tables['clients'][$id]); $n++; }
        }

            if ($g['revoked_at'] === null && $g['consented_at'] < $before) { $this->revokeGrant($id, $now); $n++; }
        }
```

```php
<?php

namespace Wonder\OAuth\Store;

use mysqli;
use RuntimeException;
use Wonder\Auth\OneTimeToken;
use Wonder\OAuth\Config;
use Wonder\Sql\Connection;

/** Persistenza mysqli. Solo prepared statement: nessun valore concatenato nell'SQL. */
final class SqlStore implements OAuthStore
{
    private const TOKEN_TABLE = ['access' => 'oauth_access_tokens', 'refresh' => 'oauth_refresh_tokens'];

    private function db(): mysqli { return Connection::Connect('main'); }

    /** @return list<array> */
    private function rows(string $sql, string $types = '', array $params = []): array
    {
        $stmt = $this->db()->prepare($sql);
        if ($stmt === false) { throw new RuntimeException('oauth_sql_prepare_failed'); }
        if ($types !== '') { $stmt->bind_param($types, ...$params); }
        $stmt->execute();
        $result = $stmt->get_result();
        $rows = $result ? $result->fetch_all(MYSQLI_ASSOC) : [];
        $stmt->close();
        return $rows;
    }

    private function exec(string $sql, string $types = '', array $params = []): int
    {
        $stmt = $this->db()->prepare($sql);
        if ($stmt === false) { throw new RuntimeException('oauth_sql_prepare_failed'); }
        if ($types !== '') { $stmt->bind_param($types, ...$params); }
        $stmt->execute();
        $affected = $stmt->affected_rows;
        $insertId = $stmt->insert_id;
        $stmt->close();
        return str_starts_with(ltrim($sql), 'INSERT') ? (int) $insertId : (int) $affected;
    }

    public function createClient(array $row): void
    {
        $this->exec('INSERT INTO `oauth_clients` (`client_id`,`client_name`,`redirect_uris`,`token_endpoint_auth_method`,`registration_ip`,`creation`) VALUES (?,?,?,?,?,?)',
            'ssssss', [$row['client_id'], $row['client_name'], $row['redirect_uris'], $row['token_endpoint_auth_method'], $row['registration_ip'], $row['creation']]);
    }
```

```php
<?php

namespace Wonder\OAuth;

use InvalidArgumentException;

/** Risorse protette dichiarate dai moduli. L'URL canonico si calcola a lettura da APP_URL. */
final class ResourceRegistry
{
    /** @var array<string,string> path => name */
    private static array $resources = [];

    public static function register(string $path, string $name): void
    {
        $path = '/'.trim($path, '/');
        if ($path === '/' || preg_match('#^/[A-Za-z0-9/_.-]+$#', $path) !== 1) {
            throw new InvalidArgumentException('oauth_resource_path_invalid');
        }
        self::$resources[$path] = trim($name) !== '' ? trim($name) : $path;
    }

    public static function all(): array
    {
        if (self::$resources === []) {
            return [];
        }
        $issuer = Config::issuer();
        $out = [];
        foreach (self::$resources as $path => $name) {
            $out[] = [
                'path' => $path,
                'name' => $name,
                'url' => $issuer.$path,
                'metadata_url' => $issuer.'/.well-known/oauth-protected-resource'.$path,
            ];
        }
        return $out;
    }

    public static function findByUrl(string $url): ?array
    {
        $url = rtrim(trim($url), '/');
        foreach (self::all() as $r) {
            if (hash_equals($r['url'], $url)) {
                return $r;
            }
        }
        return null;
    }

    public static function findByPath(string $path): ?array
    {
        $path = '/'.trim($path, '/');
        foreach (self::all() as $r) {
            if ($r['path'] === $path) {
                return $r;
            }
        }
        return null;
    }

    public static function isEmpty(): bool { return self::$resources === []; }

    public static function clear(): void { self::$resources = []; }
}
```

```php
<?php

namespace Wonder\OAuth;

use Wonder\OAuth\Store\OAuthStore;

/** Dynamic Client Registration (RFC 7591), solo client pubblici. */
final class ClientRegistry
{
    public function __construct(private readonly OAuthStore $store) {}

    public function register(array $body, string $ip): array
    {
        $limit = Config::registerRateLimit();
        if ($limit > 0 && $this->store->countClientsFromIp($ip, date('Y-m-d H:i:s', time() - 3600)) >= $limit) {
            throw new OAuthException('too_many_requests', 'Too many client registrations from this address.', 429, ['Retry-After' => '3600']);
        }

        $uris = $body['redirect_uris'] ?? null;
        if (!is_array($uris) || count($uris) < 1 || count($uris) > 10) {
            throw new OAuthException('invalid_redirect_uri', 'redirect_uris must contain 1 to 10 URIs.');
        }
        $uris = array_values($uris);
        foreach ($uris as $uri) {
            if (!is_string($uri) || !RedirectUri::isAllowed($uri)) {
                throw new OAuthException('invalid_redirect_uri', 'Redirect URIs must be https or loopback http, without fragment or userinfo.');
            }
        }

        $method = $body['token_endpoint_auth_method'] ?? 'none';
        if ($method !== 'none') {

            'client_name' => $name,
            'redirect_uris' => json_encode($uris, JSON_UNESCAPED_SLASHES),
            'token_endpoint_auth_method' => 'none',
            'registration_ip' => $ip,
            'creation' => date('Y-m-d H:i:s', $now),
        ]);

        return [
            'client_id' => $clientId,
            'client_id_issued_at' => $now,
            'client_name' => $name,
            'redirect_uris' => $uris,
            'grant_types' => array_values($grantTypes),
            'response_types' => array_values($responseTypes),
            'token_endpoint_auth_method' => 'none',
        ];
    }

    public function find(string $clientId): ?array
    {
        if ($clientId === '' || strlen($clientId) > 64) {
            return null;
        }
        $row = $this->store->findClient($clientId);
        if ($row === null || !empty($row['revoked_at'])) {
            return null;
        }
        $uris = json_decode((string) $row['redirect_uris'], true);
        $row['redirect_uris'] = is_array($uris) ? $uris : [];
        return $row;
    }

    public function touch(string $clientId): void
    {
        $this->store->touchClient($clientId, date('Y-m-d H:i:s'));
    }

    {
        if (!is_string($name)) {
            return '';
        }
        $name = strip_tags($name);
        $name = preg_replace('/[\x00-\x1F\x7F]/u', '', $name) ?? '';
        return mb_substr(trim($name), 0, 100);
    }
```

```php
<?php

namespace Wonder\OAuth;

/** Validazione di /oauth/authorize e richieste in sessione. */
final class AuthorizationRequest
{
    public static function validate(array $q, ClientRegistry $clients): array
    {
        $str = static fn (string $k): string => is_string($q[$k] ?? null) ? trim($q[$k]) : '';

        // 1. Prima del client e del redirect valido: pagina d'errore, mai redirect.
        $client = $clients->find($str('client_id'));
        $redirect = $str('redirect_uri');
        if ($client === null || $redirect === '' || !RedirectUri::matchesAny($client['redirect_uris'], $redirect)) {
            throw new OAuthException('invalid_request', 'Unknown client or unregistered redirect_uri.');
        }

        $state = $str('state') !== '' ? $str('state') : null;
        $fail = static fn (string $error, string $desc) => new OAuthException($error, $desc, 302, [], $redirect, $state);

        // 2. Da qui gli errori tornano al client.
        if ($str('response_type') !== 'code') {
            throw $fail('unsupported_response_type', 'Only response_type=code is supported.');
        }
        if ($str('code_challenge_method') !== 'S256' || !Pkce::validChallenge($str('code_challenge'))) {
            throw $fail('invalid_request', 'PKCE with S256 is required.');
        }
        $resource = ResourceRegistry::findByUrl($str('resource'));
        if ($resource === null) {
            throw $fail('invalid_target', 'Unknown or missing resource.');
        }

        return [
            'client_id' => $client['client_id'],
            'client_name' => $client['client_name'],
            'redirect_uri' => $redirect,
            'code_challenge' => $str('code_challenge'),
            'resource' => $resource['url'],
            'resource_name' => $resource['name'],
            'state' => $state,
        ];
    }

    public static function store(array $request, array &$session, ?int $now = null): string
    {
        $now ??= time();
        $requests = self::live($session, $now);
        $id = bin2hex(random_bytes(16));
        $requests[$id] = ['request' => $request, 'created' => $now];

        uasort($requests, static fn ($a, $b) => $a['created'] <=> $b['created']);
        while (count($requests) > Config::MAX_REQUESTS) {
            array_shift($requests);
        }

        $session['oauth_requests'] = $requests;
        return $id;
    }

    public static function find(array $session, string $id, ?int $now = null): ?array
    {
        if (preg_match('/^[a-f0-9]{32}$/', $id) !== 1) {
            return null;
        }
        return self::live($session, $now ?? time())[$id]['request'] ?? null;
    }

    public static function forget(array &$session, string $id): void
    {
        unset($session['oauth_requests'][$id]);
    }

    /** Logout per "Cambia account" senza perdere la richiesta: logoutUser() svuota l'intera sessione. */
    public static function switchAccount(array &$session, string $id, callable $logout): void
    {
        $request = self::find($session, $id);
        $logout();
        if ($request !== null) {
            $session['oauth_requests'][$id] = ['request' => $request, 'created' => time()];
        }
    }

    public static function redirectUrl(string $redirectUri, array $params): string
    {
        $params = array_filter($params, static fn ($v) => $v !== null && $v !== '');
        $sep = str_contains($redirectUri, '?') ? '&' : '?';
        return $redirectUri.$sep.http_build_query($params, '', '&', PHP_QUERY_RFC3986);
    }

    private static function live(array $session, int $now): array
    {
        $requests = is_array($session['oauth_requests'] ?? null) ? $session['oauth_requests'] : [];
        return array_filter($requests, static fn ($r) => is_array($r) && ($r['created'] ?? 0) + Config::REQUEST_TTL >= $now);
    }
}
```

```php
<?php

namespace Wonder\OAuth;

use Wonder\OAuth\Store\OAuthStore;

/** Codice di autorizzazione monouso (OneTimeToken purpose oauth_code, TTL 5 minuti). */
final class AuthorizationCode
{
    public function __construct(private readonly OAuthStore $store) {}

    public function issue(int $userId, array $request, int $grantId): string
    {
        return $this->store->issueCode($userId, [
            'client_id' => $request['client_id'],
            'redirect_uri' => $request['redirect_uri'],
            'code_challenge' => $request['code_challenge'],
            'resource' => $request['resource'],
            'grant_id' => $grantId,
        ], Config::CODE_TTL);
    }

    public function consume(string $code, string $clientId, string $redirectUri, ?string $resource, string $verifier): array
    {
        $record = $code !== '' ? $this->store->consumeCode($code) : null;
        $m = $record['metadata'] ?? [];

        $resourceOk = $resource === null || $resource === '' || rtrim($resource, '/') === ($m['resource'] ?? null);

        if ($record === null
            || !hash_equals((string) ($m['client_id'] ?? ''), $clientId)
            || !hash_equals((string) ($m['redirect_uri'] ?? ''), $redirectUri)
            || !$resourceOk
            || !Pkce::verify($verifier, (string) ($m['code_challenge'] ?? ''))) {
            throw new OAuthException('invalid_grant', 'Invalid, expired or already used authorization code.');
        }

        return [
            'user_id' => (int) $record['user_id'],
            'grant_id' => (int) $m['grant_id'],
            'client_id' => (string) $m['client_id'],
            'resource' => (string) $m['resource'],
        ];
    }
}
```

```php
<?php

namespace Wonder\OAuth;

use Closure;
use Wonder\OAuth\Store\OAuthStore;

/** Grant e token opachi: emissione, rotazione del refresh con rilevamento del riuso, revoca. */
final class TokenIssuer
{
    private readonly Closure $clock;

    public function __construct(private readonly OAuthStore $store, ?Closure $clock = null)
    {
        $this->clock = $clock ?? static fn (): int => time();
    }

    public function grantFor(int $userId, string $clientId, string $resource): int
    {
        $now = $this->date();
        $grant = $this->store->findActiveGrant($userId, $clientId, $resource);
        if ($grant !== null) {
            // Il consenso è stato richiesto di nuovo: la scadenza assoluta riparte da qui.
            $this->store->updateGrant((int) $grant['id'], ['consented_at' => $now]);
            return (int) $grant['id'];
        }
        return $this->store->createGrant(['user_id' => $userId, 'client_id' => $clientId, 'resource' => $resource, 'scope' => '', 'consented_at' => $now]);
    }

    public function issue(int $grantId): array
    {
        $accessTtl = Config::accessTokenTtl();
        return [
            'access_token' => $this->mint('access', 'wio_at_', $grantId, $accessTtl),
            'token_type' => 'Bearer',
            'expires_in' => $accessTtl,
            'refresh_token' => $this->mint('refresh', 'wio_rt_', $grantId, Config::refreshTokenTtl()),
        ];
    }

    public function refresh(string $refreshToken, string $clientId, ?string $resource): array
    {
        $row = $this->lookup('refresh', $refreshToken, 'wio_rt_');
        if ($row === null) {
            throw new OAuthException('invalid_grant', 'Invalid refresh token.');
        }

        $grant = $this->store->findGrant((int) $row['grant_id']);
        if ($grant === null || !hash_equals((string) $grant['client_id'], $clientId)) {
            throw new OAuthException('invalid_grant', 'Invalid refresh token.');
        }

        // Riuso: un refresh già consumato revoca l'intero grant.
        if (!empty($row['consumed_at'])) {
            $this->store->revokeGrant((int) $grant['id'], $this->date());
            throw new OAuthException('invalid_grant', 'Refresh token reuse detected; grant revoked.');
        }

        if (!empty($row['revoked_at']) || strtotime((string) $row['expires_at']) < ($this->clock)()) {
            throw new OAuthException('invalid_grant', 'Refresh token expired or revoked.');
        }

        if ($resource !== null && $resource !== '' && rtrim($resource, '/') !== $grant['resource']) {
            throw new OAuthException('invalid_target', 'Resource does not match the grant.');
        }

        $this->assertUsable((int) $grant['id']);

        if (!$this->store->consumeRefresh((int) $row['id'], $this->date())) {
            // Corsa con un'altra richiesta sullo stesso refresh: trattata come riuso.
            $this->store->revokeGrant((int) $grant['id'], $this->date());
            throw new OAuthException('invalid_grant', 'Refresh token reuse detected; grant revoked.');
        }

        return $this->issue((int) $grant['id']);
    }

    public function revokeByToken(string $token, string $clientId): void
    {
        foreach (['access' => 'wio_at_', 'refresh' => 'wio_rt_'] as $kind => $prefix) {
            if (!str_starts_with($token, $prefix)) {
                continue;
            }
            $row = $this->lookup($kind, $token, $prefix);
            $grant = $row !== null ? $this->store->findGrant((int) $row['grant_id']) : null;
            if ($grant !== null && hash_equals((string) $grant['client_id'], $clientId)) {
                $this->store->revokeGrant((int) $grant['id'], $this->date());
            }
        }
    }

    /** Grant ancora utilizzabile: non revocato, entro la scadenza assoluta, client e utente attivi. */
    public function usableGrant(int $grantId): ?array
    {
        $grant = $this->store->findGrant($grantId);
        if ($grant === null || !empty($grant['revoked_at'])
            || strtotime((string) $grant['consented_at']) + Config::grantMaxAge() < ($this->clock)()) {
            return null;
        }
        $client = $this->store->findClient((string) $grant['client_id']);
        if ($client === null || !empty($client['revoked_at'])) {
            return null;
        }
        $user = $this->store->findUser((int) $grant['user_id']);
        if ($user === null || ($user->deleted ?? false) || ($user->active ?? false) !== true) {
            return null;
        }
        return ['grant' => $grant, 'client' => $client, 'user' => $user];
    }

    /** @return array{0:string,1:string}|null */
    public static function parse(string $token, string $prefix): ?array
    {
        if (!str_starts_with($token, $prefix)) {
            return null;
        }
        $parts = explode('.', substr($token, strlen($prefix)), 2);
        if (count($parts) !== 2 || preg_match('/^[a-f0-9]{16}$/', $parts[0]) !== 1 || preg_match('/^[a-f0-9]{64}$/', $parts[1]) !== 1) {
            return null;
        }
        return [$parts[0], $parts[1]];
    }

    /** Riga del token se selector e validator corrispondono (non controlla scadenza e revoca). */
    public function lookup(string $kind, string $token, string $prefix): ?array
    {
        $parsed = self::parse($token, $prefix);
        if ($parsed === null) {
            return null;
        }
        $row = $this->store->findToken($kind, $parsed[0]);
        return ($row !== null && hash_equals((string) $row['validator_hash'], hash('sha256', $parsed[1]))) ? $row : null;
    }

    private function assertUsable(int $grantId): void
    {
        $grant = $this->store->findGrant($grantId);
        $client = $grant !== null ? $this->store->findClient((string) $grant['client_id']) : null;
        if ($client === null || !empty($client['revoked_at'])) {
            throw new OAuthException('invalid_client', 'Client not found.', 401);
        }
        if ($this->usableGrant($grantId) === null) {
            throw new OAuthException('invalid_grant', 'Grant revoked or expired, or user not active.');
        }
    }

    private function mint(string $kind, string $prefix, int $grantId, int $ttl): string
    {
        $selector = bin2hex(random_bytes(8));
        $validator = bin2hex(random_bytes(32));
        $this->store->insertToken($kind, [
            'grant_id' => $grantId,
            'selector' => $selector,
            'validator_hash' => hash('sha256', $validator),
            'expires_at' => date('Y-m-d H:i:s', ($this->clock)() + $ttl),
        ]);
        return $prefix.$selector.'.'.$validator;
    }

    private function date(): string
    {
        return date('Y-m-d H:i:s', ($this->clock)());
    }
}
```

```php
<?php

namespace Wonder\OAuth;

use Wonder\OAuth\Store\OAuthStore;

/** /oauth/token e /oauth/revoke: parametri form-urlencoded, errori RFC 6749. */
final class TokenEndpoint
{
    private readonly ClientRegistry $clients;
    private readonly TokenIssuer $issuer;
    private readonly AuthorizationCode $codes;

    public function __construct(private readonly OAuthStore $store)
    {
        $this->clients = new ClientRegistry($store);
        $this->issuer = new TokenIssuer($store);
        $this->codes = new AuthorizationCode($store);
    }

    public function token(array $post): array
    {
        $p = static fn (string $k): string => is_string($post[$k] ?? null) ? trim($post[$k]) : '';
        $clientId = $p('client_id');
        $grantType = $p('grant_type');

        if (!in_array($grantType, ['authorization_code', 'refresh_token'], true)) {
            throw new OAuthException('unsupported_grant_type', 'Supported: authorization_code, refresh_token.');
        }
        if ($this->clients->find($clientId) === null) {
            throw new OAuthException('invalid_client', 'Client not found.', 401);
        }

        $resource = $p('resource') !== '' ? $p('resource') : null;

        if ($grantType === 'refresh_token') {
            $result = $this->issuer->refresh($p('refresh_token'), $clientId, $resource);
        } else {
            $code = $this->codes->consume($p('code'), $clientId, $p('redirect_uri'), $resource, $p('code_verifier'));
            if ($this->issuer->usableGrant($code['grant_id']) === null) {
                throw new OAuthException('invalid_grant', 'Grant revoked or user not active.');
            }
            $result = $this->issuer->issue($code['grant_id']);
        }

        $this->clients->touch($clientId);
        return $result;
    }

    /** RFC 7009: risponde sempre 200, anche per token sconosciuti. */
    public function revoke(array $post): void
    {
        $token = is_string($post['token'] ?? null) ? trim($post['token']) : '';
        $clientId = is_string($post['client_id'] ?? null) ? trim($post['client_id']) : '';
        if ($token !== '' && $clientId !== '') {
            $this->issuer->revokeByToken($token, $clientId);
        }
    }
}
```

```php
<?php

namespace Wonder\OAuth;

/** Autenticazione dei bearer `wio_at_` per le risorse dei moduli. Le autorizzazioni restano alla risorsa. */
final class ResourceGuard
{
    public static function authenticate(string $resourcePath, ?array $server = null): OAuthCaller
    {
        $server ??= $_SERVER;
        $path = '/'.trim($resourcePath, '/');
        $bearer = self::bearer($server);

        if ($bearer === '') {
            throw new OAuthException('invalid_request', 'Missing bearer token.', 401, ['WWW-Authenticate' => Metadata::wwwAuthenticate($path)]);
        }

        $invalid = static fn (string $why) => new OAuthException('invalid_token', $why, 401, ['WWW-Authenticate' => Metadata::wwwAuthenticate($path, 'invalid_token', $why)]);

        $resource = ResourceRegistry::findByPath($path);
        $store = OAuth::store();
        $issuer = new TokenIssuer($store);
        $row = $issuer->lookup('access', $bearer, 'wio_at_');

        if ($resource === null || $row === null || !empty($row['revoked_at']) || strtotime((string) $row['expires_at']) < time()) {
            throw $invalid('The access token is invalid or expired.');
        }

        $usable = $issuer->usableGrant((int) $row['grant_id']);
        if ($usable === null || !hash_equals($resource['url'], (string) $usable['grant']['resource'])) {
            throw $invalid('The access token is not valid for this resource.');
        }

        $grant = $usable['grant'];
        $now = time();
        if (empty($grant['last_used_at']) || strtotime((string) $grant['last_used_at']) < $now - 60) {
            $stamp = date('Y-m-d H:i:s', $now);
            $store->updateGrant((int) $grant['id'], ['last_used_at' => $stamp]);
            $store->touchClient((string) $grant['client_id'], $stamp);
        }

        return new OAuthCaller($usable['user'], (int) $grant['user_id'], (int) $grant['id'], (string) $grant['client_id'], (string) $grant['resource']);
    }

    /** Bearer da Authorization, con fallback su REDIRECT_HTTP_AUTHORIZATION e getallheaders() (Apache CGI). */
    public static function bearer(array $server): string
    {
        $header = (string) ($server['HTTP_AUTHORIZATION'] ?? $server['REDIRECT_HTTP_AUTHORIZATION'] ?? '');

        if ($header === '' && $server === $_SERVER && function_exists('getallheaders')) {
            foreach ((array) getallheaders() as $name => $value) {
                if (strcasecmp((string) $name, 'Authorization') === 0) {
```

```php
<?php

namespace Wonder\App\Resources\OAuth;

use Wonder\App\Resources\Support\NavigationOnlyResource;
use Wonder\App\Resources\Support\NavigationSchema;
use Wonder\Http\Csrf;
use Wonder\OAuth\OAuth;
use Wonder\OAuth\RedirectUri;

/** Backend "Connessioni MCP": URL dei server MCP e grant OAuth attivi, con revoca. */
final class ConnectionsResource extends NavigationOnlyResource
{
    private const SEE_ALL = ['admin', 'administrator'];

    public static function path(): string { return 'oauth-connections'; }
    public static function icon(): string { return 'bi bi-plug'; }
    public static function titleLabel(): string { return __t('oauth.connections.title'); }

    public static function navigationSchema(): NavigationSchema
    {
        return NavigationSchema::for(static::class)
            ->inSection('dev')
            ->title(__t('oauth.connections.title'))
            ->authority([])
            ->enabled(OAuth::enabled());
    }

    public static function pageView(): string
    {
        return dirname(__DIR__, 4).'/app/view/pages/backend/oauth/connections.php';
    }

    public static function rows(?object $user): array
    {
        if ($user === null) {
            return [];
        }
        $store = OAuth::store();
        $seeAll = self::seeAll($user);
        $names = array_column(OAuth::resources(), 'name', 'url');
        $rows = [];

        foreach ($store->listActiveGrants($seeAll ? null : (int) $user->id) as $g) {
            $client = $store->findClient((string) $g['client_id']);
            $uris = json_decode((string) ($client['redirect_uris'] ?? '[]'), true) ?: [];
            $owner = $seeAll ? $store->findUser((int) $g['user_id']) : $user;
            $rows[] = [
                'grant_id' => (int) $g['id'],
                'user_name' => trim(($owner->name ?? '').' '.($owner->surname ?? '')) ?: (string) ($owner->email ?? ''),
                'client_name' => (string) ($client['client_name'] ?? $g['client_id']),
                'redirect_host' => isset($uris[0]) ? RedirectUri::host((string) $uris[0]) : '',
                'resource' => $names[$g['resource']] ?? (string) $g['resource'],
                'consented_at' => (string) $g['consented_at'],
                'last_used_at' => $g['last_used_at'] ?? null,
            ];
        }
        return $rows;
    }

    public static function seeAll(?object $user): bool
    {
        return $user !== null && array_intersect(self::SEE_ALL, (array) ($user->authority ?? [])) !== [];
    }

    public static function canRevoke(object $user, array $grant): bool
    {
        return (int) $grant['user_id'] === (int) $user->id || self::seeAll($user);
    }

    public static function handleSubmit(): never
    {
        global $USER;
        $target = __r('backend.oauth.connections');
        $grant = OAuth::store()->findGrant((int) ($_POST['grant_id'] ?? 0));

        if (!Csrf::verify() || $grant === null || !empty($grant['revoked_at']) || !is_object($USER) || !self::canRevoke($USER, $grant)) {
            \FlashAlert::custom(__t('oauth.connections.title'), __t('oauth.connections.revoke_error'), 'error');
        } else {
            OAuth::store()->revokeGrant((int) $grant['id'], date('Y-m-d H:i:s'));
            \FlashAlert::custom(__t('oauth.connections.revoked_title'), __t('oauth.connections.revoked_message'), 'success');
        }

        header('Location: '.$target, true, 303);
        exit;
    }
}
```

```php
<?php

namespace Wonder\OAuth;

use Wonder\OAuth\Store\OAuthStore;

/** Pulizia notturna: soglie fisse (7 giorni token, 30 giorni client), scadenza grant da Config. */
final class Cleanup
{
    public static function run(OAuthStore $store, int $now): array
    {
        $date = static fn (int $t): string => date('Y-m-d H:i:s', $t);

        // Prima la revoca: i grant appena revocati non tengono in vita i loro client.
        $grants = $store->revokeGrantsConsentedBefore($date($now - Config::grantMaxAge()), $date($now));

        return [
            'grants' => $grants,
            'tokens' => $store->purgeTokens($date($now - 7 * 86400)),
            'clients' => $store->purgeIdleClients($date($now - 30 * 86400)),

    }
}
```

```php
$authorize = function () use ($req, $base, $mcp, $pkce, $query, $csrf, $form, $userName, $password, &$state): array {
    [$v, $c] = $pkce();
    $r = $req('GET', $base.'/oauth/authorize?'.http_build_query(['response_type' => 'code', 'client_id' => $state['client'], 'redirect_uri' => 'http://localhost:9/callback',
        'code_challenge' => $c, 'code_challenge_method' => 'S256', 'resource' => $mcp, 'state' => 's1']));
    $loc = $r['headers']['location'] ?? '';
    if (str_contains($loc, '/oauth/login')) {
        $page = $req('GET', $loc);
        $r = $req('POST', $loc, $form(['_csrf' => $csrf($page['body']), 'username' => $userName, 'password' => $password]));
        $loc = $r['headers']['location'] ?? '';
    }
    $page = $req('GET', $loc);
    $r = $req('POST', $loc, $form(['_csrf' => $csrf($page['body']), 'action' => 'approve']));
    $q = $query($r['headers']['location'] ?? '');
    return ['code' => $q['code'] ?? '', 'verifier' => $v, 'iss' => $q['iss'] ?? '', 'state' => $q['state'] ?? ''];
};

check('due authorize paralleli: entrambi i codici validi', function () use ($authorize, $req, $base, $form, $mcp, &$state) {

    $b = $authorize();

    if ($ta['status'] !== 200 || $tb['status'] !== 200) { echo "    token: {$ta['status']} ".($ta['json']['error'] ?? '')." / {$tb['status']} ".($tb['json']['error'] ?? '')."\n"; }
    return $ta['status'] === 200 && $tb['status'] === 200 && $a['iss'] === $base && $a['state'] === 's1';
});
check('tools/list con access token', function () use ($req, $mcp, $rpc, &$state) {
    $r = $req('POST', $mcp.'/', ['headers' => ['Content-Type: application/json', 'Authorization: Bearer '.($state['tokens']['access_token'] ?? '')], 'body' => $rpc]);

});
check('refresh ruota; riuso → invalid_grant', function () use ($req, $base, $form, &$state) {
    $old = $state['tokens']['refresh_token'] ?? '';
    $n = $req('POST', $base.'/oauth/token', $form(['grant_type' => 'refresh_token', 'refresh_token' => $old, 'client_id' => $state['client']]));
    $again = $req('POST', $base.'/oauth/token', $form(['grant_type' => 'refresh_token', 'refresh_token' => $old, 'client_id' => $state['client']]));

});

    $r = $req('POST', $mcp.'/', ['headers' => ['Content-Type: application/json', 'Authorization: Bearer '.($state['tokens']['access_token'] ?? '')], 'body' => $rpc]);
    return $r['status'] === 401 && str_contains($r['headers']['www-authenticate'] ?? '', 'invalid_token');
});
```
